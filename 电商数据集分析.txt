#用户分层
def active_status(data):
    status =[]
    for i in range(len(data)):  # 用len(data)而非写死11，与实际月份数保持一致
        #若本月没有消费
        if data.iloc[i] == 0:  # 按位置取值（data[i]是按标签取，会导致KeyError）
            if len(status)>0:
                if status[i-1]=='unreg':
                    status.append('unreg')
                else:
                    status.append('unactive')
            else:
                status.append('unreg')
        #若本月消费
        else:
            if len(status)==0:
                status.append('new')
            else:
                if status[i-1]=='unactive':
                    status.append('return')
                elif status[i-1]=='unreg':
                    status.append('new')
                else:
                    status.append('active')
    return pd.Series(status,index=columns_month)

#回购率
#说明：同复购率，改用用户维度：某用户本月有消费且下月仍有消费记为回购。
def purchase_return(data):
    status = []
    for i in range(len(data)-1):
        if data.iloc[i]==1:
            if data.iloc[i+1]==1:
                status.append(1)
            else:
                status.append(0)
        else :
            status.append(np.nan)
    status.append(np.nan)
    return pd.Series(status,index=columns_month)

pivoted_purchase_user = pivoted_purchase_user.map(lambda x: 1 if x>0 else 0)
pivoted_purchase_return = pivoted_purchase_user.apply(purchase_return,axis=1)
(pivoted_purchase_return.sum()/pivoted_purchase_return.count()).plot(figsize=(12,6))
plt.xticks(range(len(columns_month)),columns_month)
plt.title('回购率(用户维度)')
