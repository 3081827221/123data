df=df.drop(columns=[c for c in df.columns if c.startswith('Unnamed')])

#============================ 数据清洗模块 ============================
# 说明：本模块对原始数据做规范化清洗，覆盖：字段类型、时间解析、缺失值、
#      重复值、异常值、文本规范化，并输出清洗前后统计报告，便于核对。
before_rows = len(df)
print('='*60)
print('数据清洗开始，清洗前记录数：%d' % before_rows)

# 1) 字段类型规范化
#    user_id/product_id/order_id 等ID为数值型但本质是标识符，统一转成字符串，
#    避免后续被当成数值参与运算或丢失精度；event_time 转为 datetime 类型。
id_cols = ['user_id', 'product_id', 'order_id', 'category_id']
for c in id_cols:
    if c in df.columns:
        df[c] = df[c].astype(str).str.replace(r'\.0$', '', regex=True)
df['event_time'] = pd.to_datetime(df['event_time'].str[:19], format="%Y-%m-%d %H:%M:%S")

# 2) 时间字段解析（截掉末尾'UTC'冗余，拆出月份/日期/星期/小时供后续分析）
df['Month'] = df['event_time'].dt.month
df['Day'] = df['event_time'].dt.day
df['Dayofweek'] = df['event_time'].dt.dayofweek
df['hour'] = df['event_time'].dt.hour

# 3) 缺失值检查与处理
#    实测：category_code 缺失12.9万条(占22.93%)、brand 缺失2.7万条(占4.83%)，
#    其余列无缺失。缺失量过大，直接删除会损失近1/4记录，
#    因此对文本列用占位符'missing'填充，保留记录并归入'缺失'类统一处理。
missing_count = df.isnull().sum()
missing_cols = missing_count[missing_count > 0].index.tolist()
print('\n[缺失值检查]')
print('各列缺失数：\n', missing_count)
if missing_cols:
    print('缺失率(%)：')
    print((df[missing_cols].isnull().mean() * 100).round(2))
    df[missing_cols] = df[missing_cols].fillna('missing')
    print('已对缺失列填充占位符 missing：', missing_cols)
else:
    print('各列均无缺失值')

# 4) 重复值检查与处理
dup_before = df.duplicated().sum()
print('\n[重复值检查] 完全重复记录数：%d' % dup_before)
df = df.drop_duplicates()
print('已删除重复记录 %d 条' % dup_before)

# 5) 异常值检查与处理
#    price 应为正数（价格<=0 视为未成交/无效订单，单独剥离），
#    age 应在合理范围(16~50)，越界的年龄视为异常并剔除。
invalid_price = (df['price'] <= 0).sum()
print('\n[异常值检查]')
print('price<=0（未成交/无效订单）条数：%d' % invalid_price)
age_valid = (df['age'] >= 16) & (df['age'] <= 50)
invalid_age = (~age_valid).sum()
print('age 超出合理区间[16,50]条数：%d' % invalid_age)
df = df[age_valid]
print('已剔除异常记录 %d 条（保留 price<=0 单独作为未成交数据）' % invalid_age)

# 6) 文本字段规范化（去首尾空格，统一为小写，避免同义不同值）
str_cols = df.select_dtypes(include=['object', 'string']).columns.tolist()
for c in str_cols:
    if c in ['event_time']:
        continue
    df[c] = df[c].astype(str).str.strip().str.lower()
print('\n[文本规范化] 已对文本列去除首尾空格并统一小写：', str_cols)

after_rows = len(df)
print('\n数据清洗完成，清洗后记录数：%d，共处理 %d 条记录' % (after_rows, before_rows - after_rows))

#用户分层
def active_status(data):
    status =[]
    for i in range(len(data)):  # 用len(data)而非写死11，与实际月份数保持一致
        #若本月没有消费
        if data.iloc[i] == 0:  # 按位置取值（data[i]是按标签取，会导致KeyError）
            if len(status)>0:
                if status[i-1]=='unreg':
                    status.append('unreg')
                else:
                    status.append('unactive')
            else:
                status.append('unreg')
        #若本月消费
        else:
            if len(status)==0:
                status.append('new')
            else:
                if status[i-1]=='unactive':
                    status.append('return')
                elif status[i-1]=='unreg':
                    status.append('new')
                else:
                    status.append('active')
    return pd.Series(status,index=columns_month)
pivoted_purchase_status = pivoted_purchase.apply(lambda x:active_status(x),axis=1)

def purchase_return(data):
    status = []
    for i in range(len(data)-1):
        if data.iloc[i]==1:
            if data.iloc[i+1]==1:
                status.append(1)
            else:
                status.append(0)
        else :
            status.append(np.nan)
    status.append(np.nan)
    return pd.Series(status,index=columns_month)

#九、相关性分析（皮尔逊相关系数）
#说明：皮尔逊相关系数 r 衡量两个连续变量之间的线性相关程度，取值范围[-1,1]。
#      |r|>=0.8 强相关；0.5<=|r|<0.8 中等相关；0.3<=|r|<0.5 弱相关；|r|<0.3 视为线性关系极弱。
#      代码运行至此，df 已被过滤为 price>0 的真实成交记录。

#9.1 用户维度：消费次数与累计消费金额的相关性（量化图11散点分布的线性强度）
user_corr = df.groupby('user_id').agg(consumption_count=('order_id','nunique'),
                                      total_amount=('price','sum'))
r_user, p_user = stats.pearsonr(user_corr['consumption_count'], user_corr['total_amount'])
print('用户消费次数与累计消费金额：皮尔逊 r=%.4f，p=%.2e（样本数=%d）' % (r_user, p_user, len(user_corr)))
plt.figure(figsize=(12,8))
sns.regplot(x='consumption_count', y='total_amount', data=user_corr,
            scatter_kws={'alpha':0.3}, line_kws={'color':'red'})
plt.xlabel('消费次数')
plt.ylabel('累计消费金额')
plt.title('用户消费次数与累计消费金额相关性（r=%.4f）' % r_user)
plt.show()
